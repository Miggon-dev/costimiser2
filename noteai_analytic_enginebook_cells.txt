def extract(
    out,
    download_artifacts,
    show_text=True,
    show_tables=False,
    show_figures=False,
    base_url="http://127.0.0.1:5000",
):
    import io

    import pandas as pd
    import plotly.graph_objects as go
    import requests
    from IPython.display import display, Markdown

    if show_text and out.get("text"):
        display(Markdown(out["text"]))

    tables = {}
    figures = {}

    if download_artifacts:
        for table in out.get("tables", []):
            table_url = base_url.rstrip("/") + table["artifact"]["url"]

            rt = requests.get(
                table_url,
                timeout=120,
            )
            rt.raise_for_status()

            df = pd.read_parquet(io.BytesIO(rt.content))
            tables[table["id"]] = df

            if show_tables:
                print(table["id"])
                display(df.head())

        for fig_payload in out.get("figures", []):
            fig_url = base_url.rstrip("/") + fig_payload["artifact"]["url"]

            rf = requests.get(
                fig_url,
                timeout=120,
            )
            rf.raise_for_status()

            fig = go.Figure(rf.json())
            figures[fig_payload["id"]] = fig

            if show_figures:
                fig.show()

    else:
        for table in out.get("tables", []):
            df = pd.DataFrame(table["inline"]["data"])
            tables[table["id"]] = df

            if show_tables:
                print(table["id"])
                display(df.head())

        for fig_payload in out.get("figures", []):
            fig_json = fig_payload["plotly_json"]

            fig = go.Figure(fig_json)
            figures[fig_payload["id"]] = fig

            if show_figures:
                fig.show()

    return tables, figures


def wait_for_job(
    job_id: str,
    endpoint_url: str = "http://127.0.0.1:5000/",
    poll_interval: float = 2.0,
    timeout: float = 15 * 60,
):
    import time
    import requests

    base_url = endpoint_url.rstrip("/")
    job_url = f"{base_url}/jobs/{job_id}"

    start_time = time.monotonic()
    previous_status = None

    while True:
        r = requests.get(
            job_url,
            timeout=30,
        )

        print("Job HTTP status:", r.status_code)

        if not r.ok:
            try:
                response = r.json()
            except ValueError:
                response = r.text

            raise RuntimeError(
                f"Could not retrieve job {job_id}: {response}"
            )

        job = r.json()
        status = job.get("status")

        if status != previous_status:
            print(f"Job status: {status}")
            previous_status = status

        if status == "completed":
            if "result" not in job:
                raise RuntimeError(
                    f"Job {job_id} completed but returned no result."
                )

            return job["result"]

        if status == "failed":
            error = job.get("error", "Unknown optimisation error")
            result = job.get("result")

            if result:
                raise RuntimeError(
                    f"Optimisation failed: {error}\n"
                    f"Engine response: {result}"
                )

            raise RuntimeError(
                f"Optimisation failed: {error}"
            )

        if status not in {"queued", "running"}:
            raise RuntimeError(
                f"Unexpected job status: {status}"
            )

        elapsed = time.monotonic() - start_time

        if elapsed >= timeout:
            raise TimeoutError(
                f"Job {job_id} did not finish within "
                f"{timeout:.0f} seconds."
            )

        time.sleep(poll_interval)


def run_query(
    query: str,
    download_artifacts: bool = True,
    endpoint_url: str = "http://127.0.0.1:5000/ask-card",
):
    import requests

    r = requests.post(
        endpoint_url,
        json={
            "query": query,
            "download_artifacts": download_artifacts,
            "diagnosis_summary": True,
            "cost_driver_summary": False,
        },
        timeout=120,
    )

    print("HTTP status:", r.status_code)

    try:
        response = r.json()
    except ValueError:
        print("The server did not return JSON:")
        print(r.text)
        return None, None

    if r.ok:
        base_url = endpoint_url.rsplit("/", 1)[0]

        return extract(
            response,
            download_artifacts,
            show_tables=True,
            show_figures=True,
            base_url=base_url,
        )

    print("\nERROR RESPONSE")
    print(response)

    if isinstance(response, dict):
        print("\nMessage:")
        print(response.get("message", response))

    return None, None

def ask_engine(
    params: dict,
    endpoint: str,
    download_artifacts: bool = True,
    endpoint_url: str = "http://127.0.0.1:5000/",
    run_async: bool = True,
    poll_interval: float = 5.0,
    job_timeout: float = 30 * 60,
):
    import io
    import time

    import pandas as pd
    import requests

    base_url = endpoint_url.rstrip("/")
    endpoint = endpoint.lstrip("/")
    full_endpoint = f"{base_url}/{endpoint}"

    # Do not modify the original dictionary.
    request_params = dict(params)

    request_params["download_artifacts"] = download_artifacts
    request_params["async"] = run_async

    print(f"Calling: {full_endpoint}")
    print(f"Mode: {'asynchronous' if run_async else 'synchronous'}")

    # Endpoint returning parquet directly.
    if endpoint == "process-data/parquet":
        try:
            response = requests.get(
                full_endpoint,
                params=request_params,
                timeout=(10, 120),
            )

        except requests.exceptions.RequestException as exc:
            print(f"Request failed: {exc}")
            return None, None

        print("HTTP status:", response.status_code)

        if response.ok:
            process_df = pd.read_parquet(
                io.BytesIO(response.content)
            )
            return {"process": process_df}, None

        print("\nERROR RESPONSE")

        try:
            error_response = response.json()
        except ValueError:
            error_response = response.text

        print(error_response)
        return None, None

    # Normal JSON POST endpoints.
    submission_started = time.monotonic()

    try:
        response = requests.post(
            full_endpoint,
            json=request_params,
            timeout=(10, 30 if run_async else job_timeout),
        )

    except requests.exceptions.ReadTimeout:
        elapsed = time.monotonic() - submission_started

        if run_async:
            print(
                f"The initial POST timed out after {elapsed:.1f} seconds.\n"
                "An asynchronous endpoint should return a job_id "
                "almost immediately. Check that the service was "
                "restarted and that this endpoint supports async mode."
            )
        else:
            print(
                f"The synchronous request timed out after "
                f"{elapsed:.1f} seconds."
            )

        return None, None

    except requests.exceptions.ConnectionError as exc:
        print(f"Could not connect to the engine: {exc}")
        return None, None

    except requests.exceptions.RequestException as exc:
        print(f"Request failed: {exc}")
        return None, None

    submission_elapsed = time.monotonic() - submission_started

    print(
        f"Submission response received after "
        f"{submission_elapsed:.2f} seconds."
    )
    print("HTTP status:", response.status_code)

    try:
        response_payload = response.json()

    except ValueError:
        print("\nThe server did not return valid JSON:")
        print(response.text)
        return None, None

    if not response.ok:
        print("\nERROR RESPONSE")
        print(response_payload)

        if isinstance(response_payload, dict):
            print("\nMessage:")
            print(
                response_payload.get(
                    "message",
                    response_payload.get(
                        "text",
                        response_payload,
                    ),
                )
            )

        return None, None

    # Generic asynchronous response handling.
    if response.status_code == 202:
        if "job_id" not in response_payload:
            print(
                "\nThe endpoint returned HTTP 202 but no job_id."
            )
            print(response_payload)
            return None, None

        job_id = response_payload["job_id"]
        job_type = response_payload.get(
            "job_type",
            endpoint,
        )

        print(f"{job_type} accepted.")
        print(f"Job ID: {job_id}")

        try:
            response_payload = wait_for_job(
                job_id=job_id,
                endpoint_url=endpoint_url,
                poll_interval=poll_interval,
                timeout=job_timeout,
            )

        except Exception as exc:
            print(f"\nJob failed: {exc}")
            return None, None

        print(f"{job_type} completed.")

    # Defensive check: async=True was requested, but the endpoint
    # returned a final synchronous response.
    elif run_async:
        print(
            "Warning: asynchronous mode was requested, but the "
            f"endpoint returned HTTP {response.status_code} instead "
            "of HTTP 202. Treating the response as a final result."
        )

    return extract(
        response_payload,
        download_artifacts,
        show_tables=True,
        show_figures=True,
        base_url=base_url,
    )

def fibre_cost_from_row(row) -> float:    
    #require_columns(row, fibre_features(), "fibre")    
    if type(row) == pd.Series:
        row = row.to_dict()
    basis_weight = row["Current_basis_weight"]
    starch_uptake = row["Starch_uptake_by_paper_Bottom_Roll__g/m2_"] + row["Starch_uptake_by_paper_Top_Roll__g/m2_"]
    moisture = row["Current_reel_moisture_average(reel)"]

    return 146.46* (basis_weight * (1 - moisture / 100) - starch_uptake) / basis_weight


def fibre_cost(X):
    """
    Behaves like the model-based cost functions:
    - accepts one row: pd.Series / dict
    - accepts a dataframe: pd.DataFrame
    """    
    if isinstance(X, list):
        X = pd.DataFrame(X)
    if isinstance(X, pd.DataFrame):
        
        missing = [c for c in fibre_features() if c not in X.columns]
        if missing:
            raise KeyError(f"Missing columns for 'fibre': {missing}")

        basis_weight = X["Current_basis_weight"]
        starch_uptake = X["Starch_uptake_by_paper_Bottom_Roll__g/m2_"] + X["Starch_uptake_by_paper_Top_Roll__g/m2_"]
        moisture = X["Current_reel_moisture_average(reel)"]

        return 146.46  * (basis_weight * (1 - moisture / 100) - starch_uptake) / basis_weight    
    return float(fibre_cost_from_row(X))

def starch_features():
